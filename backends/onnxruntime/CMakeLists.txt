if(NOT MIPK_ONNXRUNTIME_ROOT)
    if(DEFINED ENV{ONNXRUNTIME_ROOT})
        set(MIPK_ONNXRUNTIME_ROOT "$ENV{ONNXRUNTIME_ROOT}")
    else()
        if(WIN32)
            set(_ort_platform win-x64)
        else()
            set(_ort_platform linux-x64)
        endif()
        set(_ort_bundled "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/onnxruntime/${_ort_platform}")
        if(EXISTS "${_ort_bundled}/include/onnxruntime_c_api.h")
            set(MIPK_ONNXRUNTIME_ROOT "${_ort_bundled}")
        endif()
    endif()
endif()
find_path(MIPK_ONNXRUNTIME_INCLUDE onnxruntime_c_api.h
    HINTS "${MIPK_ONNXRUNTIME_ROOT}/include" /usr/include/onnxruntime /usr/local/include/onnxruntime)
find_library(MIPK_ONNXRUNTIME_LIBRARY onnxruntime
    HINTS "${MIPK_ONNXRUNTIME_ROOT}/lib" /usr/lib /usr/local/lib)
if(NOT MIPK_ONNXRUNTIME_INCLUDE OR NOT MIPK_ONNXRUNTIME_LIBRARY)
    message(FATAL_ERROR "onnxruntime not found; run scripts/fetch_onnxruntime.sh (scripts\\fetch_onnxruntime.ps1 on Windows) or set MIPK_ONNXRUNTIME_ROOT")
endif()
if(WIN32)
    # Executables need onnxruntime.dll beside them; mipk_copy_runtime_dlls()
    # copies these after each build.
    file(GLOB _ort_dlls "${MIPK_ONNXRUNTIME_ROOT}/lib/*.dll")
    set_property(GLOBAL APPEND PROPERTY MIPK_RUNTIME_DLLS ${_ort_dlls})
endif()
add_library(mipk_onnxruntime_sdk INTERFACE)
target_include_directories(mipk_onnxruntime_sdk INTERFACE "${MIPK_ONNXRUNTIME_INCLUDE}")
target_link_libraries(mipk_onnxruntime_sdk INTERFACE "${MIPK_ONNXRUNTIME_LIBRARY}")
get_filename_component(_ort_lib_dir "${MIPK_ONNXRUNTIME_LIBRARY}" DIRECTORY)
set(MIPK_ONNXRUNTIME_LIBRARY_DIR "${_ort_lib_dir}" CACHE INTERNAL "")
mipk_add_module(SOURCES mipk_onnxruntime.c LINK mipk_onnxruntime_sdk)
set_target_properties(mipk_onnxruntime PROPERTIES BUILD_RPATH "${_ort_lib_dir}")
