add_executable(mipk_run mipk_run.c)
target_link_libraries(mipk_run PRIVATE mipk::mipk)
target_include_directories(mipk_run PRIVATE "${PROJECT_SOURCE_DIR}/third_party/stb")
target_compile_options(mipk_run PRIVATE ${MIPK_WARNING_FLAGS})
if(NOT MSVC)
    target_compile_options(mipk_run PRIVATE -Wno-unused-function)
endif()
mipk_copy_runtime_dlls(mipk_run)
if(MIPK_ONNXRUNTIME_LIBRARY_DIR)
    set_target_properties(mipk_run PROPERTIES BUILD_RPATH "${MIPK_ONNXRUNTIME_LIBRARY_DIR}")
endif()
mipk_write_capabilities(TARGET mipk_run OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/mipk_capabilities.json")
